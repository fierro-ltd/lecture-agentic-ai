```python
import asyncio
import logging

logger = logging.getLogger(__name__)

async def get_temporal_client() -> TemporalClient:
    global _temporal_client
    if _temporal_client is None:
        delays = [5, 10, 20]
        last_err = None
        for attempt, delay in enumerate(delays, 1):
            try:
                _temporal_client = await TemporalClient.connect(settings.TEMPORAL_ADDRESS)
                logger.info("Connected to Temporal at %s", settings.TEMPORAL_ADDRESS)
                return _temporal_client
            except Exception as e:
                last_err = e
                logger.warning(
                    "Temporal connection attempt %d/%d failed: %s. Retrying in %ds...",
                    attempt, len(delays), e, delay,
                )
                await asyncio.sleep(delay)
        raise RuntimeError(f"Failed to connect to Temporal after {len(delays)} attempts") from last_err
    return _temporal_client
```

- [ ] **Step 2: Write the seeding script**

Create `scripts/seed-agent-instructions.py`:

```python
#!/usr/bin/env python3
"""Seed Paperclip agent instructions from SOUL.md files.

Connects directly to the Paperclip PostgreSQL database and updates
agent instruction fields to match hermes/agents/*/SOUL.md content.
Idempotent — safe to run multiple times.
"""

import os
import sys
import glob
import time
import psycopg2


def get_db_connection(max_retries=10, delay=3):
    """Connect to Paperclip database with retries (waits for Paperclip to init schema)."""
    db_url = os.getenv(
        "PAPERCLIP_DATABASE_URL",
        "postgresql://lecture:lecture_dev@postgres:5432/paperclip",
    )
    for attempt in range(1, max_retries + 1):
        try:
            conn = psycopg2.connect(db_url)
            return conn
        except psycopg2.OperationalError as e:
            print(f"[seed] DB connection attempt {attempt}/{max_retries}: {e}")
            time.sleep(delay)
    print("[seed] ERROR: Could not connect to Paperclip database")
    sys.exit(1)


def discover_schema(conn):
    """Find the table and column that stores agent instructions."""
    with conn.cursor() as cur:
        # Check for 'member' table with 'instructions' column (Paperclip's schema)
        cur.execute("""
            SELECT column_name FROM information_schema.columns
            WHERE table_name = 'member' AND column_name IN ('instructions', 'system_prompt', 'soul')
        """)
        row = cur.fetchone()
        if row:
            return "member", row[0], "slug"

        # Fallback: check 'agent' table
        cur.execute("""
            SELECT column_name FROM information_schema.columns
            WHERE table_name = 'agent' AND column_name IN ('instructions', 'system_prompt', 'soul')
        """)
        row = cur.fetchone()
        if row:
            return "agent", row[0], "slug"

    print("[seed] ERROR: Could not find agent instructions column in Paperclip schema")
    sys.exit(1)


def load_soul_files(agents_dir):
    """Load all SOUL.md files, keyed by agent slug."""
    souls = {}
    for path in glob.glob(os.path.join(agents_dir, "*/SOUL.md")):
        slug = os.path.basename(os.path.dirname(path))
        with open(path, "r") as f:
            souls[slug] = f.read()
        print(f"[seed] Loaded SOUL.md for agent: {slug}")
    return souls


def find_agent_name_column(conn, table):
    """Find the column used for agent name/slug matching."""
    with conn.cursor() as cur:
        cur.execute("""
            SELECT column_name FROM information_schema.columns
            WHERE table_name = %s AND column_name IN ('slug', 'name', 'identifier')
            ORDER BY CASE column_name WHEN 'slug' THEN 1 WHEN 'identifier' THEN 2 WHEN 'name' THEN 3 END
        """, (table,))
        row = cur.fetchone()
        return row[0] if row else "name"


def seed_instructions(conn, table, instructions_col, souls):
    """Update agent instructions in the database."""
    name_col = find_agent_name_column(conn, table)
    updated = 0

    with conn.cursor() as cur:
        # Get all agents from the database
        cur.execute(f"SELECT id, {name_col} FROM {table}")
        agents = cur.fetchall()

        for agent_id, agent_name in agents:
            # Try matching by slug convention: "Assessment Quality Agent" -> "assessment-quality"
            # or direct slug match
            agent_slug = agent_name.lower().replace(" ", "-").replace("_", "-")
            # Remove common suffixes for matching
            for suffix in ["-agent", "-director"]:
                slug_without_suffix = agent_slug.replace(suffix, "")
                if slug_without_suffix in souls:
                    agent_slug = slug_without_suffix
                    break

            if agent_slug in souls:
                cur.execute(
                    f"UPDATE {table} SET {instructions_col} = %s WHERE id = %s",
                    (souls[agent_slug], agent_id),
                )
                updated += 1
                print(f"[seed] Updated instructions for: {agent_name} (matched: {agent_slug})")
            else:
                print(f"[seed] No SOUL.md found for: {agent_name} (tried: {agent_slug})")

    conn.commit()
    return updated


def main():
    agents_dir = os.getenv("AGENTS_DIR", "/agents")

    print("[seed] Loading SOUL.md files...")
    souls = load_soul_files(agents_dir)
    if not souls:
        print("[seed] No SOUL.md files found. Nothing to do.")
        return

    print("[seed] Connecting to Paperclip database...")
    conn = get_db_connection()

    try:
        print("[seed] Discovering schema...")
        table, instructions_col, _ = discover_schema(conn)
        print(f"[seed] Found: table={table}, column={instructions_col}")

        print("[seed] Seeding instructions...")
        updated = seed_instructions(conn, table, instructions_col, souls)
        print(f"[seed] Done. Updated {updated}/{len(souls)} agents.")
    finally:
        conn.close()


if __name__ == "__main__":
    main()
```

"""Webhook notification activity for review events."""

from __future__ import annotations

import httpx
from temporalio import activity

from src.config import settings


@activity.defn
async def send_notification(payload: dict) -> bool:
    """POST a JSON payload to the configured webhook URL. No-op if URL is not set."""
    url = settings.NOTIFICATION_WEBHOOK_URL
    if not url:
        return False

    try:
        async with httpx.AsyncClient(timeout=30) as client:
            response = await client.post(url, json=payload)
            response.raise_for_status()
        return True
    except Exception as e:
        activity.logger.warning("Notification webhook failed: %s", e)
        return False
```

# Demo: Approve a submission via the HAST API.

    \"decision\": \"$DECISION\",
    \"reviewer_notes\": \"Reviewed via demo script. Decision: $DECISION.\",

  }")

"""Structured JSON logging configuration for HAST services."""

import logging
import sys
from pythonjsonlogger import json as json_log


def setup_logging(service_name: str):
    """Configure structured JSON logging for a HAST service."""
    handler = logging.StreamHandler(sys.stdout)
    formatter = json_log.JsonFormatter(
        fmt="%(asctime)s %(levelname)s %(name)s %(message)s",
        rename_fields={"asctime": "timestamp", "levelname": "level", "name": "logger"},
        static_fields={"service": service_name},
    )
    handler.setFormatter(formatter)

    root = logging.getLogger()
    root.handlers.clear()
    root.addHandler(handler)
    root.setLevel(logging.INFO)
```

- [ ] **Step 3: Add correlation ID middleware to api.py**

Add middleware that reads `X-Correlation-ID` header (or generates one) and makes it available via context var:

```python
import uuid
from contextvars import ContextVar
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request

correlation_id_var: ContextVar[str] = ContextVar("correlation_id", default="")

class CorrelationIDMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        cid = request.headers.get("X-Correlation-ID", str(uuid.uuid4()))
        correlation_id_var.set(cid)
        response = await call_next(request)
        response.headers["X-Correlation-ID"] = cid
        return response
```

Add `app.add_middleware(CorrelationIDMiddleware)` before the CORS middleware.
